```bash
# 예: 2019년 '연령(구간)' 항목의 응답값
curl "$BASE/getServeyCareerBreakDataList?serviceKey=$PUBLIC_DATA_ENCODED_API_KEY&pageNo=1&numOfRows=100&type=json&fctExmnYr=2019&fctExmnTrgtSn=1&fctExmnItmSn=9"
# → {"fctExmnDataSn":1,"fctExmnTiemVl":"6","fctExmnCdNm":"6. 만50-54세"}, ...
```
